num_attributes = len(data[0]) - 1
hypothesis = ['0'] * num_attributes
first = True
print("\nUpdating hypothesis: \n")

for row in data:
    if row[-1] == "Yes":
      if first:
        hypothesis = row[:-1]
        first = False
      else:
        for i in range(num_attributes):
          if hypothesis[i] != row[i]:
            hypothesis[i] = "?"
      print(hypothesis)
