from sklearn.linear_model import LinearRegression  
from sklearn.cluster import KMeans 
import numpy as np 
 
# 1. Supervised Learning (Linear Regression)  
X_sup = np.array([[1], [2], [3], [4]]) 
y_sup = np.array([2, 4, 6, 8]) 
reg = LinearRegression().fit(X_sup, y_sup) 
print("Supervised Prediction for [5]:", reg.predict([[5]])[0]) 
 
# 2. Unsupervised Learning (K-Means) 
X_unsup = np.array([[1, 2], [1, 4], [1, 0], [10, 2], [10, 4], [10, 0]]) 
kmeans = KMeans(n_clusters=2, random_state=0, n_init='auto').fit(X_unsup) 
 
print("Unsupervised Cluster Labels:", kmeans.labels_)
